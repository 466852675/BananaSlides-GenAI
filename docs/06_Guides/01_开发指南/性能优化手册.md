### 2. 缓存策略

```typescript
// 配置缓存
const configCache = new Map<string, {
  config: any;
  timestamp: number;
}>();
const CACHE_TTL = 60000; // 1分钟

// 使用缓存
async function getCachedConfig(key: string) {
  const cached = configCache.get(key);
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return cached.config;
  }
  
  const config = await fetchConfigFromDB(key);
  configCache.set(key, { config, timestamp: Date.now() });
  return config;
}

// Redis 缓存（生产环境）
import Redis from 'ioredis';
const redis = new Redis();

async function getUserWithCache(userId: string) {
  const cached = await redis.get(`user:${userId}`);
  if (cached) return JSON.parse(cached);
  
  const user = await prisma.user.findUnique({ where: { id: userId } });
  await redis.setex(`user:${userId}`, 300, JSON.stringify(user));
  return user;
}
```

## AI 服务优化

### 1. 请求合并

```typescript
// 批量生成请求
async function batchGenerate(prompts: string[]) {
  const batchSize = 5;
  const results = [];
  
  for (let i = 0; i < prompts.length; i += batchSize) {
    const batch = prompts.slice(i, i + batchSize);
    const batchResults = await Promise.all(
      batch.map(p => generateWithRetry(p))
    );
    results.push(...batchResults);
  }
  
  return results;
}
```

### 2. 超时与重试

```typescript
async function generateWithRetry(
  prompt: string, 
  maxRetries: number = 3
): Promise<string> {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await Promise.race([
        callAIService(prompt),
        new Promise((_, reject) => 
          setTimeout(() => reject(new Error('Timeout')), 30000)
        )
      ]);
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      await delay(1000 * Math.pow(2, i)); // 指数退避
    }
  }
}
```
