# 🚨 YH-AI PPT 错误码规范

> **文档状态**: ✅ 已发布  
> **最后同步代码版本**: v1.2.3 (2026-02-16)  

## 错误响应格式

```json
{
  "success": false,
  "error": {
    "code": "ERROR_CODE",
    "message": "用户友好的错误描述",
    "details": {          // 可选，详细错误信息
      "field": "email",
      "issue": "invalid_format"
    }
  }
}
```

## 错误码结构

错误码采用 `CATEGORY_SPECIFIC` 格式：

| 类别前缀 | 说明 | 示例 |
|---------|------|------|
| `AUTH_` | 认证相关 | `AUTH_UNAUTHORIZED` |
| `USER_` | 用户相关 | `USER_NOT_FOUND` |
| `VALIDATION_` | 参数验证 | `VALIDATION_REQUIRED` |
| `BUSINESS_` | 业务逻辑 | `BUSINESS_INSUFFICIENT_POINTS` |
| `SYSTEM_` | 系统错误 | `SYSTEM_ERROR` |
| `RATE_` | 限流相关 | `RATE_LIMIT_EXCEEDED` |
| `AI_` | AI生成相关 | `AI_GENERATION_FAILED` |
| `PAYMENT_` | 支付相关 | `PAYMENT_FAILED` |
| `REFUND_` | 退款相关 | `REFUND_NOT_ELIGIBLE` |

## 错误处理最佳实践

### 前端处理

```typescript
// 统一错误处理
async function apiCall(url: string, options: RequestInit) {
    try {
        const response = await fetch(url, options);
        const data = await response.json();
        
        if (!data.success) {
            handleApiError(data.error);
            throw new ApiError(data.error);
        }
        
        return data.data;
    } catch (error) {
        if (error instanceof ApiError) {
            throw error;
        }
        // 网络错误等
        throw new ApiError({
            code: 'SYSTEM_ERROR',
            message: '网络错误，请检查连接'
        });
    }
}

function handleApiError(error: ApiError) {
    switch (error.code) {
        case 'AUTH_UNAUTHORIZED':
            // 跳转到登录
            window.location.href = '/login';
            break;
            
        case 'BUSINESS_INSUFFICIENT_POINTS':
            // 显示充值弹窗
            showRechargeModal();
            break;
            
        case 'RATE_LIMIT_EXCEEDED':
            // 显示友好的提示
            toast.error('操作太频繁，请稍后再试');
            break;
            
        case 'AI_GENERATION_FAILED':
            // 提供重试选项
            showRetryDialog(error.message);
            break;
            
        default:
            // 通用错误提示
            toast.error(error.message || '操作失败，请重试');
    }
}
```

### 后端实现

```typescript
// 统一错误类
class AppError extends Error {
    constructor(
        public code: string,
        message: string,
        public statusCode: number = 500,
        public details?: Record<string, any>
    ) {
        super(message);
    }
}

// 错误示例
class ValidationError extends AppError {
    constructor(message: string, details?: Record<string, any>) {
        super('VALIDATION_REQUIRED', message, 400, details);
    }
}

class InsufficientPointsError extends AppError {
    constructor(currentPoints: number, requiredPoints: number) {
        super(
            'BUSINESS_INSUFFICIENT_POINTS',
            `积分不足，当前余额 ${currentPoints}，需要 ${requiredPoints}`,
            402,
            { currentPoints, requiredPoints }
        );
    }
}

// 错误中间件
app.use((err: Error, req: Request, res: Response, next: NextFunction) => {
    if (err instanceof AppError) {
        res.status(err.statusCode).json({
            success: false,
            error: {
                code: err.code,
                message: err.message,
                details: err.details
            }
        });
    } else {
        console.error('Unhandled error:', err);
        res.status(500).json({
            success: false,
            error: {
                code: 'SYSTEM_ERROR',
                message: '系统错误，请稍后重试'
            }
        });
    }
});
```
