### 1.1 当前权限控制机制

| 层级 | 实现方式 | 覆盖范围 | 状态 |
|------|---------|---------|------|

| **角色检查** | `requireAdmin()`, `requireSuperAdmin()` | 管理后台 | ✅ 正常 |

```

```

### 1.3 核心问题

**问题 1：权限保存但未生效**
- 用户在角色管理页面勾选/取消权限
- 数据正确保存到 `RolePermission` 表
- 但 API 调用时**没有检查具体权限**
- 任何 `ADMIN` 角色用户都能访问所有管理功能

**问题 2：缺少细粒度权限中间件**
```typescript
// 当前仅有角色级检查
router.delete('/:id', authenticate, requireAdmin, deleteLead);

// 缺少权限级检查
router.delete('/:id', 
    authenticate, 
    requireAdmin,
    requirePermission('admin.leads.delete'),  // ❌ 不存在
    deleteLead
);
```

**问题 3：前端无权限控制组件**
- 按钮默认显示，根据角色显示/隐藏
- 没有基于具体权限的动态控制
- 用户界面看到的功能和实际权限不一致

#### 3.1.1 权限校验中间件

**文件**: `server/src/middlewares/auth.middleware.ts`

```typescript
// ============================================
// 新增：权限校验中间件

/**
 * 检查用户是否具有指定权限
 * @param permissionCode 权限代码 (如 admin.leads.delete)
 */
export const requirePermission = (permissionCode: string) => {
    return async (
        req: Request,
        res: Response,
        next: NextFunction
    ): Promise<void> => {
        try {
            // 1. 检查登录状态
            if (!req.user) {
                return res.status(401).json({
                    success: false,
                    error: { 
                        code: 'UNAUTHORIZED', 
                        message: '请先登录' 
                    }
                });
            }

            // 2. 超级管理员拥有所有权限
            if (req.user.role === UserRole.SUPER_ADMIN) {
                next();
                return;
            }

            // 3. 查询用户角色的所有权限
            const userPermissions = await prisma.rolePermission.findMany({
                where: { role: req.user.role },
                include: { permission: true }
            });

            // 4. 检查是否有所需权限
            const hasPermission = userPermissions.some(
                rp => rp.permission.code === permissionCode
            );

            if (!hasPermission) {
                return res.status(403).json({
                    success: false,
                    error: { 
                        code: 'FORBIDDEN',
                        message: `权限不足，需要权限: ${permissionCode}`,
                        requiredPermission: permissionCode,
                        userRole: req.user.role
                    }
                });
            }

            // 5. 通过校验
            next();

        } catch (error) {
            console.error('[PermissionCheck] Error:', error);
            return res.status(500).json({
                success: false,
                error: { code: 'PERMISSION_CHECK_ERROR', message: '权限校验失败' }
            });
        }
    };
};

/**
 * 检查用户是否具有任一权限（OR 逻辑）
 */
export const requireAnyPermission = (...permissionCodes: string[]) => {
    return async (
        req: Request,
        res: Response,
        next: NextFunction
    ): Promise<void> => {
        try {
            if (!req.user) {
                return res.status(401).json({
                    success: false,
                    error: { code: 'UNAUTHORIZED', message: '请先登录' }
                });
            }

            // 超级管理员直接通过
            if (req.user.role === UserRole.SUPER_ADMIN) {
                next();
                return;
            }

            const userPermissions = await prisma.rolePermission.findMany({
                where: { role: req.user.role },
                include: { permission: true }
            });

            // 检查是否有任一权限
            const hasAnyPermission = permissionCodes.some(code =>
                userPermissions.some(rp => rp.permission.code === code)
            );

            if (!hasAnyPermission) {
                return res.status(403).json({
                    success: false,
                    error: { 
                        code: 'FORBIDDEN',
                        message: `权限不足，需要以下任一权限: ${permissionCodes.join(', ')}`
                    }
                });
            }

            next();

        } catch (error) {
            console.error('[PermissionCheck] Error:', error);
            return res.status(500).json({
                success: false,
                error: { code: 'PERMISSION_CHECK_ERROR', message: '权限校验失败' }
            });
        }
    };
};

/**
 * 数据范围控制（可选扩展）
 * own: 只能看自己的数据
 * all: 可以看全部数据
 */
export const requireDataScope = (scope: 'own' | 'all' = 'own') => {
    return (
        req: Request,
        res: Response,
        next: NextFunction
    ): void => {
        // 在 req 中附加数据范围要求
        req.dataScope = scope;
        next();
    };
};
```

```typescript
/**
 * 获取用户列表（支持数据范围控制）
 * @param userId 当前用户ID
 * @param scope 数据范围 'own' | 'dept' | 'all'
 */
export async function listUsers(userId: string, scope: string = 'all') {
    const currentUser = await prisma.user.findUnique({
        where: { id: userId },
        include: { rolePermissions: { include: { permission: true } } }
    });

    let whereClause: any = {};

    // 检查是否有查看所有用户的权限
    const canViewAllUsers = currentUser?.rolePermissions.some(
        rp => rp.permission.code === 'admin.users.view.all'
    );

    if (!canViewAllUsers) {
        // 只能看本部门或自己的
        whereClause = { id: userId };
    }

    return prisma.user.findMany({ where: whereClause });
}
```

#### 3.2.1 权限 Hook

**文件**: `src/hooks/usePermissions.ts` (新建)

```typescript
import { useQuery } from '@tanstack/react-query';
import { apiClient } from '../api/client';

interface PermissionHookReturn {
    hasPermission: (code: string) => boolean;
    hasAnyPermission: (codes: string[]) => boolean;
    isLoading: boolean;
}

export function usePermissions(): PermissionHookReturn {
    const { data: permissions = [], isLoading } = useQuery({
        queryKey: ['admin', 'permissions'],
        queryFn: async () => {
            const res = await apiClient.get('/admin/permissions');
            return res.data;
        }
    });

    const { data: rolePermissions = [] } = useQuery({
        queryKey: ['admin', 'roles', 'my-permissions'],
        queryFn: async () => {
            // 获取当前用户的角色权限
            const res = await apiClient.get('/admin/roles/my-permissions');
            return res.data;
        }
    });

    const hasPermission = (code: string): boolean => {
        return rolePermissions.some((p: any) => p.permission?.code === code);
    };

    const hasAnyPermission = (codes: string[]): boolean => {
        return codes.some(code => hasPermission(code));
    };

    return { hasPermission, hasAnyPermission, permissions, isLoading };
}

export default usePermissions;
```

#### 3.2.2 权限守卫组件

**文件**: `src/components/PermissionGuard.tsx` (新建)

```typescript
import React from 'react';
import { usePermissions } from '../hooks/usePermissions';

interface PermissionGuardProps {
    permission?: string;
    permissions?: string[]; // OR 逻辑：满足任一即可
    requireAll?: boolean;    // AND 逻辑：需要所有权限
    children: React.ReactNode;
    fallback?: React.ReactNode; // 无权限时的展示
    hideOnMissing?: boolean;  // 无权限时完全隐藏
}

export const PermissionGuard: React.FC<PermissionGuardProps> = ({
    permission,
    permissions,
    requireAll = false,
    children,
    fallback = null,
    hideOnMissing = false
}) => {
    const { hasPermission, hasAnyPermission, isLoading } = usePermissions();

    // 加载中
    if (isLoading) return <>{children}</>;

    // 单个权限检查
    if (permission) {
        const has = hasPermission(permission);
        if (!has) {
            return hideOnMissing ? null : (
                <div className="opacity-50 cursor-not-allowed relative">
                    {children}
                    {fallback}
                </div>
            );
        }
    }

    // 多个权限检查
    if (permissions && permissions.length > 0) {
        const has = requireAll
            ? permissions.every(code => hasPermission(code))  // AND
            : permissions.some(code => hasPermission(code)); // OR

        if (!has) {
            return hideOnMissing ? null : (
                <div className="opacity-50 cursor-not-allowed relative">
                    {children}
                    {fallback}
                </div>
            );
        }
    }

    return <>{children}</>;
};

export default PermissionGuard;
```

#### 3.2.3 权限提示组件

**文件**: `src/components/PermissionTooltip.tsx` (新建)

```typescript
import React from 'react';
import { Lock } from 'lucide-react';

interface PermissionTooltipProps {
    requiredPermission: string;
    children: React.ReactNode;
    className?: string;
}

export const PermissionTooltip: React.FC<PermissionTooltipProps> = ({
    requiredPermission,
    children,
    className = ''
}) => {
    const { hasPermission } = usePermissions();

    const hasAccess = hasPermission(requiredPermission);

    if (hasAccess) {
        return <>{children}</>;
    }

    return (
        <div className={`relative group ${className}`}>
            {children}
            
            {/* 遮罩层 */}
            <div className="absolute inset-0 bg-white/60 backdrop-blur-[2px] rounded-lg flex items-center justify-center gap-2 cursor-not-allowed">
                <Lock size={16} className="text-slate-500" />
                <span className="text-xs font-bold text-slate-600">
                    无权限
                </span>
            </div>

            {/* 提示 */}
            <div className="absolute top-full left-0 mt-2 hidden group-hover:block z-50">
                <div className="px-3 py-2 bg-slate-900 text-white text-xs rounded-lg shadow-xl whitespace-nowrap">
                    需要 <code className="bg-slate-700 px-1 rounded">{requiredPermission}</code> 权限
                </div>
            </div>
        </div>
    );
};

export default PermissionTooltip;
```

### 7.2 关键代码片段

**权限查询优化（缓存）**:
```typescript
// 权限缓存
const permissionCache = new Map<string, Set<string>>();

export async function checkPermission(userId: string, permissionCode: string): Promise<boolean> {
    const cacheKey = `${userId}:${permissionCode}`;
    
    // 检查缓存
    if (permissionCache.has(cacheKey)) {
        return permissionCache.get(cacheKey) === true;
    }
    
    // 查询数据库
    const hasPermission = await queryPermissionFromDB(userId, permissionCode);
    
    // 写入缓存（5分钟过期）
    permissionCache.set(cacheKey, hasPermission);
    setTimeout(() => permissionCache.delete(cacheKey), 5 * 60 * 1000);
    
    return hasPermission;
}
```
