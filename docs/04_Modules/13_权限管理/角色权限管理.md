# 角色权限管理

## 概述

角色权限管理模块提供系统角色定义、权限分配、访问控制的完整管理能力。管理员可以通过可视化界面配置不同角色的权限矩阵，实现细粒度的访问控制。系统支持7种内置角色和自定义权限分配，满足企业级权限管理需求。

## 角色体系

### 角色定义 (Prisma Enum)

```prisma
// server/prisma/schema.prisma

enum UserRole {
  USER          // 普通用户 - 基础功能
  BASIC         // 基础会员 - VIP 1级
  PROFESSIONAL  // 专业会员 - VIP 2级
  PREMIUM       // 高级会员 - VIP 3级
  ENTERPRISE    // 企业会员 - VIP 4级
  ADMIN         // 管理员 - 后台管理
  SUPER_ADMIN   // 超级管理员 - 全系统控制
}
```

### 角色层级与能力

| 角色 | 等级 | 基础能力 | 管理权限 | 典型用户 |
|------|------|----------|----------|----------|
| USER | 1 | PPT生成、项目管理(自己的) | 无 | 免费用户 |
| BASIC | 2 | 标准积分额度、基础模板 | 无 | 轻度付费 |
| PROFESSIONAL | 3 | 中等积分、高级模板、生图 | 无 | 专业创作者 |
| PREMIUM | 4 | 高积分、优先队列、全功能 | 无 | 重度用户 |
| ENTERPRISE | 5 | 团队功能、API访问、无限积分 | 团队成员管理 | 企业客户 |
| ADMIN | 6 | 所有用户功能 | 用户/订单/内容管理 | 运营人员 |
| SUPER_ADMIN | 7 | 所有功能 | 全系统配置、权限分配 | 系统管理员 |

## 数据库模型

### 权限表

```prisma
model Permission {
  id              String           @id @default(uuid())
  code            String           @unique  // 权限唯一标识
  name            String                      // 权限名称
  module          String                      // 所属模块
  description     String?                     // 权限描述
  sortOrder       Int              @default(0)
  createdAt       DateTime         @default(now())
  rolePermissions RolePermission[]
}
```

### 角色权限关联表

```prisma
model RolePermission {
  id           String     @id @default(uuid())
  role         UserRole            // 角色枚举值
  permissionId String              // 权限ID
  createdAt    DateTime   @default(now())
  permission   Permission @relation(fields: [permissionId], references: [id], onDelete: Cascade)

  @@unique([role, permissionId])   // 角色+权限联合唯一
}
```

## 核心服务实现

### 权限服务

```typescript
// server/src/services/permission.service.ts

import { prisma } from '../db';
import { UserRole } from '@prisma/client';

export class PermissionService {
  /**
   * 获取角色的所有权限
   */
  static async getRolePermissions(role: UserRole) {
    const permissions = await prisma.rolePermission.findMany({
      where: { role },
      include: { permission: true }
    });
    
    return permissions.map(rp => rp.permission);
  }

  /**
   * 检查角色是否有特定权限
   */
  static async hasPermission(
    role: UserRole, 
    permissionCode: string
  ): Promise<boolean> {
    // 超级管理员拥有所有权限
    if (role === UserRole.SUPER_ADMIN) {
      return true;
    }
    
    const rolePermission = await prisma.rolePermission.findFirst({
      where: {
        role,
        permission: { code: permissionCode }
      }
    });
    
    return !!rolePermission;
  }

  /**
   * 为角色分配权限
   */
  static async assignPermission(
    role: UserRole, 
    permissionCode: string
  ) {
    const permission = await prisma.permission.findUnique({
      where: { code: permissionCode }
    });
    
    if (!permission) {
      throw new Error('权限不存在');
    }
    
    return await prisma.rolePermission.upsert({
      where: {
        role_permissionId: {
          role,
          permissionId: permission.id
        }
      },
      create: {
        role,
        permissionId: permission.id
      },
      update: {}
    });
  }

  /**
   * 批量分配权限给角色
   */
  static async batchAssignPermissions(
    role: UserRole,
    permissionCodes: string[]
  ) {
    const results = [];
    
    for (const code of permissionCodes) {
      try {
        const result = await this.assignPermission(role, code);
        results.push({ code, success: true, result });
      } catch (error: any) {
        results.push({ code, success: false, error: error.message });
      }
    }
    
    return results;
  }

  /**
   * 移除角色的权限
   */
  static async revokePermission(
    role: UserRole, 
    permissionCode: string
  ) {
    const permission = await prisma.permission.findUnique({
      where: { code: permissionCode }
    });
    
    if (!permission) return;
    
    await prisma.rolePermission.deleteMany({
      where: {
        role,
        permissionId: permission.id
      }
    });
  }

  /**
   * 获取所有权限列表（按模块分组）
   */
  static async getAllPermissions() {
    const permissions = await prisma.permission.findMany({
      orderBy: [
        { module: 'asc' },
        { sortOrder: 'asc' }
      ]
    });
    
    // 按模块分组
    const grouped = permissions.reduce((acc, perm) => {
      if (!acc[perm.module]) {
        acc[perm.module] = [];
      }
      acc[perm.module].push(perm);
      return acc;
    }, {} as Record<string, typeof permissions>);
    
    return grouped;
  }

  /**
   * 创建新权限
   */
  static async createPermission(data: {
    code: string;
    name: string;
    module: string;
    description?: string;
    sortOrder?: number;
  }) {
    return await prisma.permission.create({
      data: {
        code: data.code,
        name: data.name,
        module: data.module,
        description: data.description,
        sortOrder: data.sortOrder ?? 0
      }
    });
  }

  /**
   * 删除权限
   */
  static async deletePermission(permissionCode: string) {
    return await prisma.permission.delete({
      where: { code: permissionCode }
    });
  }

  /**
   * 获取所有角色的权限矩阵
   */
  static async getPermissionMatrix() {
    const roles = Object.values(UserRole);
    const permissions = await this.getAllPermissions();
    
    const matrix: Record<string, Record<string, boolean>> = {};
    
    for (const [module, perms] of Object.entries(permissions)) {
      matrix[module] = {};
      for (const perm of perms) {
        matrix[module][perm.code] = {} as any;
        for (const role of roles) {
          const hasPerm = await this.hasPermission(role, perm.code);
          (matrix[module][perm.code] as any)[role] = hasPerm;
        }
      }
    }
    
    return matrix;
  }
}
```

## 前端组件设计

### 角色权限矩阵组件

```tsx
// PermissionMatrix.tsx
import React, { useState, useEffect } from 'react';
import { useMutation, useQuery } from '@tanstack/react-query';
import { Check, X, Loader2 } from 'lucide-react';
import { toast } from 'react-hot-toast';

const USER_ROLES = [
  { value: 'USER', label: '普通用户', color: 'gray' },
  { value: 'BASIC', label: '基础会员', color: 'blue' },
  { value: 'PROFESSIONAL', label: '专业会员', color: 'indigo' },
  { value: 'PREMIUM', label: '高级会员', color: 'purple' },
  { value: 'ENTERPRISE', label: '企业会员', color: 'pink' },
  { value: 'ADMIN', label: '管理员', color: 'orange' },
  { value: 'SUPER_ADMIN', label: '超级管理员', color: 'red' }
];

interface Permission {
  id: string;
  code: string;
  name: string;
  module: string;
  description?: string;
}

interface PermissionMatrixProps {
  readOnly?: boolean;
}

export const PermissionMatrix: React.FC<PermissionMatrixProps> = ({
  readOnly = false
}) => {
  const [localMatrix, setLocalMatrix] = useState<Record<string, Record<string, boolean>>>({});
  
  const { data: permissions, isLoading } = useQuery({
    queryKey: ['permissions'],
    queryFn: fetchPermissions
  });
  
  const { data: matrix } = useQuery({
    queryKey: ['permission-matrix'],
    queryFn: fetchPermissionMatrix
  });

  const updateMutation = useMutation({
    mutationFn: updateRolePermission,
    onSuccess: () => {
      toast.success('权限更新成功');
    },
    onError: () => {
      toast.error('权限更新失败');
    }
  });

  useEffect(() => {
    if (matrix) {
      setLocalMatrix(matrix);
    }
  }, [matrix]);

  const handleToggle = (role: string, permissionCode: string) => {
    if (readOnly) return;
    
    const newValue = !localMatrix[permissionCode]?.[role];
    
    setLocalMatrix(prev => ({
      ...prev,
      [permissionCode]: {
        ...prev[permissionCode],
        [role]: newValue
      }
    }));
    
    updateMutation.mutate({ role, permissionCode, granted: newValue });
  };

  if (isLoading) {
    return (
      <div className="flex items-center justify-center h-64">
        <Loader2 className="w-8 h-8 animate-spin text-blue-500" />
      </div>
    );
  }

  // 按模块分组显示
  const groupedPermissions = permissions?.reduce((acc, perm) => {
    if (!acc[perm.module]) acc[perm.module] = [];
    acc[perm.module].push(perm);
    return acc;
  }, {} as Record<string, Permission[]>);

  return (
    <div className="space-y-8">
      {Object.entries(groupedPermissions || {}).map(([module, perms]) => (
        <div key={module} className="bg-white rounded-lg shadow overflow-hidden">
          <div className="px-6 py-4 bg-gray-50 border-b">
            <h3 className="text-lg font-semibold">{module}</h3>
          </div>
          
          <div className="overflow-x-auto">
            <table className="w-full">
              <thead>
                <tr className="bg-gray-50">
                  <th className="px-6 py-3 text-left text-sm font-medium text-gray-500">
                    权限
                  </th>
                  {USER_ROLES.map(role => (
                    <th
                      key={role.value}
                      className="px-4 py-3 text-center text-xs font-medium"
                    >
                      <div className={`text-${role.color}-600`}>
                        {role.label}
                      </div>
                    </th>
                  ))}
                </tr>
              </thead>
              <tbody className="divide-y">
                {perms.map(perm => (
                  <tr key={perm.code} className="hover:bg-gray-50">
                    <td className="px-6 py-4">
                      <div className="text-sm font-medium">{perm.name}</div>
                      <div className="text-xs text-gray-500">{perm.code}</div>
                      {perm.description && (
                        <div className="text-xs text-gray-400 mt-1">
                          {perm.description}
                        </div>
                      )}
                    </td>
                    {USER_ROLES.map(role => {
                      const hasPermission = localMatrix[perm.code]?.[role.value] ?? false;
                      const isSuperAdmin = role.value === 'SUPER_ADMIN';
                      
                      return (
                        <td key={role.value} className="px-4 py-4 text-center">
                          <button
                            onClick={() => handleToggle(role.value, perm.code)}
                            disabled={readOnly || isSuperAdmin || updateMutation.isPending}
                            className={`inline-flex items-center justify-center w-8 h-8 rounded transition-colors ${
                              isSuperAdmin
                                ? 'bg-green-100 text-green-600 cursor-default'
                                : hasPermission
                                ? 'bg-green-100 text-green-600 hover:bg-green-200'
                                : 'bg-gray-100 text-gray-400 hover:bg-gray-200'
                            }`}
                          >
                            {hasPermission ? (
                              <Check className="w-4 h-4" />
                            ) : (
                              <X className="w-4 h-4" />
                            )}
                          </button>
                        </td>
                      );
                    })}
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      ))}
    </div>
  );
};
```

## 使用示例

### 在路由中使用权限控制

```typescript
// server/src/routes/admin.routes.ts

import { Router } from 'express';
import { authenticate, requireAdmin, requirePermission } from '../middlewares/auth.middleware';

const router = Router();

// 基础认证和管理员角色检查
router.use(authenticate, requireAdmin);

// 用户管理路由
router.get('/users', 
  requirePermission('admin.users.view'), 
  listUsers
);

router.put('/users/:id', 
  requirePermission('admin.users.manage'), 
  updateUser
);

router.delete('/users/:id', 
  requirePermission('admin.users.delete'), 
  deleteUser
);

// 退款审核路由
router.put('/refunds/:id/approve', 
  requirePermission('admin.refund.approve'), 
  approveRefund
);

router.put('/refunds/:id/approve-second', 
  requirePermission('admin.refund.approve.second'), 
  secondApproveRefund
);
```
