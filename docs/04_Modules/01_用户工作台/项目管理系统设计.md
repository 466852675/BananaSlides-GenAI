# 项目管理系统设计

## 概述

项目管理系统是 YH-AI PPT 平台的**核心数据载体**，负责演示文稿项目的全生命周期管理。系统支持**项目-幻灯片两级结构**，提供大纲编辑、样式管理、版本控制等完整功能。

## 核心功能特性

| 功能特性 | 说明 | 技术亮点 |
|---------|------|---------|
| 项目CRUD | 创建、查询、更新、删除项目 | 软删除、配额检查 |
| 幻灯片管理 | 批量同步幻灯片数据 | JSON序列化、事务保证 |
| 项目置顶 | 支持项目置顶排序 | 按isPinned+updatedAt排序 |
| 场景类型 | BUSINESS/ACADEMIC/CREATIVE | 影响AI生成策略 |
| 配额管理 | VIP等级决定项目数量上限 | 实时检查 |
| 版本快照 | 支持项目版本历史 | ProjectSnapshot模型 |

## 数据库模型

### 1. 项目表 (Project)

```prisma
model Project {
  id            String            @id @default(uuid())
  displayId     String?                             // 展示ID: PID-YYMMDD-XXXXXX
  
  // 基础信息
  title         String                              // 项目标题
  status        String            @default("idle")  // 项目状态
  isPinned      Boolean           @default(false)   // 是否置顶
  scenarioType  String            @default("BUSINESS") // 场景类型
  
  // 配置数据 (JSON序列化)
  globalConfig  String                              // 大纲JSON
  styleMap      String?                             // 风格图Map JSON
  thumbnailUrl  String?                             // 缩略图URL
  
  // 时间戳
  createdAt     DateTime          @default(now())
  updatedAt     DateTime          @default(now())
  completedAt   DateTime?                           // 完成时间
  
  // 关联
  userId        String?
  user          User?             @relation(fields: [userId], references: [id])
  snapshots     ProjectSnapshot[]                   // 版本快照
  items         Slide[]                             // 幻灯片列表
  
  @@index([userId])
  @@index([createdAt])
}
```

### 2. 幻灯片表 (Slide)

```prisma
model Slide {
  id              String   @id @default(uuid())
  projectId       String
  
  // 基础信息
  index           Int                             // 页面序号 (0-based)
  pageType        String                          // cover/directory/transition/content/end
  contentType     String                          // 内容类型
  title           String                          // 页面标题
  content         String                          // 详细内容 (Markdown)
  brief           String?                         // 内容摘要
  
  // 变体图片 (JSON序列化)
  variants        String   @default("[]")         // 变体图片数组JSON
  variantCount    Int      @default(2)            // 变体数量
  previewUrl      String?                         // 预览图URL
  originalFileRef String?                         // 原始文件引用
  
  // 状态
  status          String   @default("idle")       // idle/pending/completed
  createdAt       DateTime @default(now())
  
  // 关联
  project         Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
}
```

## 核心算法实现

### 1. 项目创建与配额检查

```typescript
// server/src/services/project.service.ts

export async function createProject(
  userId: string,
  data: CreateProjectInput
): Promise<Project> {
  // 1. 检查项目配额
  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: { vipLevel: true, vipExpiresAt: true }
  });

  const quota = getProjectQuota(user?.vipLevel || 0);
  const currentCount = await prisma.project.count({
    where: { 
      userId,
      status: { not: 'deleted' }
    }
  });

  if (currentCount >= quota) {
    throw new Error(`项目数量已达上限(${quota})，请升级VIP或删除旧项目`);
  }

  // 2. 生成展示ID
  const displayId = generateDisplayId();

  // 3. 创建项目
  const project = await prisma.project.create({
    data: {
      displayId,
      title: data.title,
      scenarioType: data.scenarioType || 'BUSINESS',
      globalConfig: JSON.stringify(data.globalConfig || {}),
      styleMap: data.styleMap ? JSON.stringify(data.styleMap) : null,
      userId
    }
  });

  // 4. 如果有大纲，批量创建幻灯片
  if (data.outline && data.outline.length > 0) {
    await syncSlides(project.id, data.outline);
  }

  return project;
}

// 生成展示ID: PID-YYMMDD-XXXXXX
function generateDisplayId(): string {
  const date = new Date();
  const dateStr = date.toISOString().slice(2, 10).replace(/-/g, '');
  const random = Math.floor(Math.random() * 1000000).toString().padStart(6, '0');
  return `PID-${dateStr}-${random}`;
}

// 项目配额表
function getProjectQuota(vipLevel: number): number {
  const quotas = {
    0: 10,    // 普通用户
    1: 30,    // Bronze
    2: 50,    // Silver
    3: Infinity, // Gold+
    4: Infinity,
    5: Infinity
  };
  return quotas[vipLevel] || 10;
}
```

### 2. 幻灯片批量同步

```typescript
export async function syncSlides(
  projectId: string,
  slides: SlideInput[]
): Promise<void> {
  await prisma.$transaction(async (tx) => {
    // 1. 获取现有幻灯片
    const existingSlides = await tx.slide.findMany({
      where: { projectId },
      select: { id: true, index: true }
    });

    const existingMap = new Map(existingSlides.map(s => [s.index, s.id]));
    const newIndices = new Set(slides.map(s => s.index));

    // 2. 删除不存在的幻灯片
    for (const existing of existingSlides) {
      if (!newIndices.has(existing.index)) {
        await tx.slide.delete({ where: { id: existing.id } });
      }
    }

    // 3. 创建或更新幻灯片
    for (const slide of slides) {
      const slideData = {
        projectId,
        index: slide.index,
        pageType: slide.pageType,
        contentType: slide.contentType || 'text',
        title: slide.title,
        content: slide.content || '',
        brief: slide.brief,
        variants: JSON.stringify(slide.variants || []),
        variantCount: slide.variantCount || 2,
        previewUrl: slide.previewUrl,
        status: slide.status || 'idle'
      };

      const existingId = existingMap.get(slide.index);
      if (existingId) {
        // 更新
        await tx.slide.update({
          where: { id: existingId },
          data: slideData
        });
      } else {
        // 创建
        await tx.slide.create({ data: slideData });
      }
    }
  });
}
```

### 3. 项目列表查询

```typescript
export async function getProjectList(
  userId: string,
  options: {
    page?: number;
    limit?: number;
    search?: string;
    scenarioType?: string;
  }
): Promise<{
  items: Project[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
  }
}> {
  const { page = 1, limit = 10, search, scenarioType } = options;

  const where: any = {
    userId,
    status: { not: 'deleted' }
  };

  if (search) {
    where.title = { contains: search };
  }

  if (scenarioType) {
    where.scenarioType = scenarioType;
  }

  const [items, total] = await Promise.all([
    prisma.project.findMany({
      where,
      orderBy: [
        { isPinned: 'desc' },    // 置顶优先
        { updatedAt: 'desc' }     // 最近更新优先
      ],
      skip: (page - 1) * limit,
      take: limit,
      include: {
        items: {
          select: {
            id: true,
            index: true,
            pageType: true,
            status: true,
            previewUrl: true
          }
        }
      }
    }),
    prisma.project.count({ where })
  ]);

  return {
    items,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit)
    }
  };
}
```

### 4. 项目快照创建

```typescript
export async function createSnapshot(
  projectId: string,
  summary?: string
): Promise<ProjectSnapshot> {
  const project = await prisma.project.findUnique({
    where: { id: projectId },
    include: { items: true }
  });

  if (!project) throw new Error('项目不存在');

  // 获取当前最大版本号
  const lastSnapshot = await prisma.projectSnapshot.findFirst({
    where: { projectId },
    orderBy: { version: 'desc' }
  });

  const nextVersion = (lastSnapshot?.version || 0) + 1;

  // 创建快照
  const snapshot = await prisma.projectSnapshot.create({
    data: {
      projectId,
      version: nextVersion,
      summary: summary || `版本 ${nextVersion}`,
      data: JSON.stringify({
        title: project.title,
        scenarioType: project.scenarioType,
        globalConfig: project.globalConfig,
        styleMap: project.styleMap,
        slides: project.items
      })
    }
  });

  return snapshot;
}

// 从快照恢复
export async function restoreFromSnapshot(
  projectId: string,
  snapshotId: string
): Promise<Project> {
  const snapshot = await prisma.projectSnapshot.findUnique({
    where: { id: snapshotId }
  });

  if (!snapshot || snapshot.projectId !== projectId) {
    throw new Error('快照不存在');
  }

  const data = JSON.parse(snapshot.data);

  // 更新项目
  const project = await prisma.project.update({
    where: { id: projectId },
    data: {
      title: data.title,
      scenarioType: data.scenarioType,
      globalConfig: data.globalConfig,
      styleMap: data.styleMap
    }
  });

  // 恢复幻灯片
  if (data.slides && data.slides.length > 0) {
    await syncSlides(projectId, data.slides);
  }

  return project;
}
```

interface ProjectCardProps {
  project: {
    id: string;
    displayId: string;
    title: string;
    isPinned: boolean;

      previewUrl?: string;

export const ProjectCard: React.FC<ProjectCardProps> = ({
  project,
  onPin,
  onDelete
}) => {
  return (
    <div className="project-card">
      {project.isPinned && <span className="pin-badge">置顶</span>}
      
      <div className="thumbnail">
        {project.thumbnailUrl ? (
          <img src={project.thumbnailUrl} alt={project.title} />
        ) : (
          <div className="placeholder">
            {project.items.slice(0, 3).map((item, i) => (
              <img key={i} src={item.previewUrl} alt="" />
            ))}
          </div>
        )}
      </div>
      
      <h3>{project.title}</h3>
      <p>{project.displayId}</p>
      
      <div className="actions">
        <button onClick={() => onPin(project.id)}>
          {project.isPinned ? '取消置顶' : '置顶'}
        </button>
        <button onClick={() => onDelete(project.id)}>删除</button>
      </div>
    </div>
  );
};
