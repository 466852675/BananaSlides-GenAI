# 快照版本历史

## 概述

快照版本历史是 YH-AI PPT 的**项目版本控制系统**，为用户提供"时间机器"功能，记录项目每次保存的完整状态。系统支持**AI自动生成变更摘要**、**版本对比**和**一键恢复**，确保用户创作过程的安全性和可追溯性。

## 核心功能特性

| 功能特性 | 说明 | 技术亮点 |
|---------|------|---------|
| 版本快照 | 保存项目完整状态 | JSON序列化 |
| AI摘要 | 自动生成变更描述 | AI文本生成 |
| 版本恢复 | 回滚到任意历史版本 | 数据替换 |
| 版本对比 | 查看版本间差异 | 文本diff |
| 保留策略 | 最多保留50个版本 | 自动清理 |
| 手动+自动 | 支持手动和自动保存 | 定时任务 |

## 数据库模型

### 1. 项目快照表 (ProjectSnapshot)

```prisma
model ProjectSnapshot {
  id        String   @id @default(uuid())
  projectId String
  
  version   Int                             // 版本号（自增）
  summary   String?                         // AI生成的变更摘要
  data      String                          // 快照数据（JSON序列化）
  
  createdAt DateTime @default(now())
  
  project   Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  
  @@index([projectId])
  @@index([createdAt])
}
```

### 2. 快照数据结构

```typescript
interface SnapshotData {
  // 项目基本信息
  project: {
    id: string;
    title: string;
    scenarioType: string;
    globalConfig: StyleConfig;
    styleMap: StyleMap;
  };
  
  // 幻灯片列表
  slides: Array<{
    id: string;
    index: number;
    pageType: string;
    title: string;
    content: string;
    brief?: string;
    variants: string[];
    previewUrl?: string;
  }>;
  
  // 元数据
  metadata: {
    slideCount: number;
    wordCount: number;
    imageCount: number;
  };
}
```

## 核心算法实现

### 1. 创建快照

```typescript
// server/src/services/snapshot.service.ts

export async function createSnapshot(
  projectId: string,
  summary?: string
): Promise<ProjectSnapshot> {
  const project = await prisma.project.findUnique({
    where: { id: projectId },
    include: { items: true }
  });
  
  if (!project) throw new Error('项目不存在');
  
  // 获取当前最大版本号
  const lastSnapshot = await prisma.projectSnapshot.findFirst({
    where: { projectId },
    orderBy: { version: 'desc' }
  });
  
  const nextVersion = (lastSnapshot?.version || 0) + 1;
  
  // 构建快照数据
  const snapshotData: SnapshotData = {
    project: {
      id: project.id,
      title: project.title,
      scenarioType: project.scenarioType,
      globalConfig: JSON.parse(project.globalConfig),
      styleMap: project.styleMap ? JSON.parse(project.styleMap) : null
    },
    slides: project.items.map(slide => ({
      id: slide.id,
      index: slide.index,
      pageType: slide.pageType,
      title: slide.title,
      content: slide.content,
      brief: slide.brief,
      variants: JSON.parse(slide.variants),
      previewUrl: slide.previewUrl
    })),
    metadata: {
      slideCount: project.items.length,
      wordCount: project.items.reduce((sum, s) => sum + s.content.length, 0),
      imageCount: project.items.reduce((sum, s) => sum + JSON.parse(s.variants).length, 0)
    }
  };
  
  // 如果没有提供摘要，使用AI生成
  let finalSummary = summary;
  if (!finalSummary) {
    finalSummary = await generateSnapshotSummary(project, lastSnapshot);
  }
  
  // 创建快照
  const snapshot = await prisma.projectSnapshot.create({
    data: {
      projectId,
      version: nextVersion,
      summary: finalSummary,
      data: JSON.stringify(snapshotData)
    }
  });
  
  // 检查并清理旧版本（保留最近50个）
  await cleanupOldSnapshots(projectId, 50);
  
  return snapshot;
}
```

### 2. AI生成变更摘要

```typescript
import { AIService } from './ai.service';

async function generateSnapshotSummary(
  currentProject: Project,
  lastSnapshot?: ProjectSnapshot
): Promise<string> {
  // 如果没有上一个快照，返回初始版本
  if (!lastSnapshot) {
    return `创建项目: ${currentProject.title}`;
  }
  
  const lastData: SnapshotData = JSON.parse(lastSnapshot.data);
  
  // 构建变更上下文
  const changes: string[] = [];
  
  // 检查标题变更
  if (lastData.project.title !== currentProject.title) {
    changes.push(`标题改为"${currentProject.title}"`);
  }
  
  // 检查幻灯片数量变化
  const slideDiff = currentProject.items.length - lastData.slides.length;
  if (slideDiff > 0) {
    changes.push(`添加${slideDiff}页幻灯片`);
  } else if (slideDiff < 0) {
    changes.push(`删除${Math.abs(slideDiff)}页幻灯片`);
  }
  
  // 如果有明显变更，使用AI生成更智能的摘要
  if (changes.length === 0 || changes.length > 2) {
    const diffContext = buildDiffContext(lastData, currentProject);
    
    try {
      const aiSummary = await AIService.generateSnapshotSummary(diffContext);
      return aiSummary || changes.join(', ') || '常规保存';
    } catch {
      return changes.join(', ') || '常规保存';
    }
  }
  
  return changes.join(', ');
}

function buildDiffContext(lastData: SnapshotData, current: Project): string {
  const parts: string[] = [];
  
  // 比较幻灯片内容变化
  current.items.forEach((slide, idx) => {
    const lastSlide = lastData.slides.find(s => s.index === slide.index);
    if (lastSlide) {
      if (lastSlide.title !== slide.title) {
        parts.push(`第${idx + 1}页标题: "${lastSlide.title}" → "${slide.title}"`);
      }
      if (lastSlide.content !== slide.content) {
        parts.push(`第${idx + 1}页内容更新`);
      }
    }
  });
  
  return parts.join('\n') || '无显著变更';
}
```

### 3. 版本恢复

```typescript
export async function restoreSnapshot(
  projectId: string,
  snapshotId: string
): Promise<Project> {
  const snapshot = await prisma.projectSnapshot.findUnique({
    where: { id: snapshotId }
  });
  
  if (!snapshot || snapshot.projectId !== projectId) {
    throw new Error('快照不存在');
  }
  
  const data: SnapshotData = JSON.parse(snapshot.data);
  
  // 先创建当前状态的备份（防止误操作）
  await createSnapshot(projectId, `恢复前备份 (目标: 版本${snapshot.version})`);
  
  // 更新项目数据
  const updatedProject = await prisma.project.update({
    where: { id: projectId },
    data: {
      title: data.project.title,
      scenarioType: data.project.scenarioType,
      globalConfig: JSON.stringify(data.project.globalConfig),
      styleMap: data.project.styleMap ? JSON.stringify(data.project.styleMap) : null
    }
  });
  
  // 删除现有幻灯片
  await prisma.slide.deleteMany({
    where: { projectId }
  });
  
  // 恢复快照中的幻灯片
  await prisma.slide.createMany({
    data: data.slides.map(slide => ({
      projectId,
      index: slide.index,
      pageType: slide.pageType,
      title: slide.title,
      content: slide.content,
      brief: slide.brief,
      variants: JSON.stringify(slide.variants),
      previewUrl: slide.previewUrl
    }))
  });
  
  return updatedProject;
}
```

### 4. 清理旧版本

```typescript
async function cleanupOldSnapshots(
  projectId: string,
  keepCount: number = 50
): Promise<void> {
  const snapshots = await prisma.projectSnapshot.findMany({
    where: { projectId },
    orderBy: { version: 'desc' },
    skip: keepCount
  });
  
  if (snapshots.length > 0) {
    await prisma.projectSnapshot.deleteMany({
      where: {
        id: { in: snapshots.map(s => s.id) }
      }
    });
    
    console.log(`[Snapshot] 清理了 ${snapshots.length} 个旧版本`);
  }
}
```

## API 接口

### 获取版本历史

**Endpoint:** `GET /api/projects/:id/snapshots?page=1&limit=10`

**Response:**
```typescript
{
  success: true;
  data: {
    items: Array<{
      id: string;
      version: number;
      summary: string;
      createdAt: string;
      metadata: {
        slideCount: number;
        wordCount: number;
      };
    }>;
    pagination: {
      page: number;
      limit: number;
      total: number;
    }
  }
}
```

### 创建快照

**Endpoint:** `POST /api/projects/:id/snapshots`

**Request Body:**
```typescript
{
  summary?: string;  // 可选，不提供则AI自动生成
}
```

### 恢复快照

**Endpoint:** `POST /api/projects/:id/snapshots/:snapshotId/restore`

**Response:**
```typescript
{
  success: true;
  message: '已恢复到版本 5';
}
```

### 获取快照详情

**Endpoint:** `GET /api/projects/:id/snapshots/:snapshotId`

**Response:**
```typescript

  success: true;
```

## 前端集成

### 1. 版本历史组件

```typescript
// src/components/VersionHistory.tsx

import { useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { formatDistanceToNow } from 'date-fns';

interface VersionHistoryProps {
  projectId: string;
}

export const VersionHistory: React.FC<VersionHistoryProps> = ({ projectId }) => {
  const [selectedVersion, setSelectedVersion] = useState<string | null>(null);
  
  const { data: snapshots } = useQuery({
    queryKey: ['project', projectId, 'snapshots'],
    queryFn: async () => {
      const res = await fetch(`/api/projects/${projectId}/snapshots`);
      return res.json();
    }
  });
  
  const restoreMutation = useMutation({
    mutationFn: async (snapshotId: string) => {
      const res = await fetch(
        `/api/projects/${projectId}/snapshots/${snapshotId}/restore`,
        { method: 'POST' }
      );
      return res.json();
    }
  });
  
  return (
    <div className="version-history">
      <h3>版本历史</h3>
      
      <div className="snapshot-list">
        {snapshots?.data?.items.map((snapshot: any) => (
          <div
            key={snapshot.id}
            className={`snapshot-item ${selectedVersion === snapshot.id ? 'selected' : ''}`}
            onClick={() => setSelectedVersion(snapshot.id)}
          >
            <div className="version-badge">v{snapshot.version}</div>
            <div className="snapshot-info">
              <p className="summary">{snapshot.summary}</p>
              <span className="time">
                {formatDistanceToNow(new Date(snapshot.createdAt), { addSuffix: true })}
              </span>
              <span className="meta">
                {snapshot.metadata.slideCount}页 · {snapshot.metadata.wordCount}字
              </span>
            </div>
            <button
              onClick={(e) => {
                e.stopPropagation();
                if (confirm(`确定恢复到版本 ${snapshot.version} 吗？`)) {
                  restoreMutation.mutate(snapshot.id);
                }
              }}
            >
              恢复
            </button>
          </div>
        ))}
      </div>
    </div>
  );
};
```

### 2. 保存按钮（带快照功能）

```typescript
// src/components/SaveButton.tsx

import { useState } from 'react';
import { useMutation } from '@tanstack/react-query';

export const SaveButton: React.FC<{
  projectId: string;
  onSave: () => Promise<void>;
}> = ({ projectId, onSave }) => {
  const [showSummaryInput, setShowSummaryInput] = useState(false);
  const [summary, setSummary] = useState('');
  
  const saveMutation = useMutation({
    mutationFn: async () => {
      // 1. 保存项目
      await onSave();
      
      // 2. 创建快照
      await fetch(`/api/projects/${projectId}/snapshots`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ summary: summary || undefined })
      });
    }
  });
  
  return (
    <div className="save-button-group">
      <button
        onClick={() => saveMutation.mutate()}
        disabled={saveMutation.isPending}
      >
        {saveMutation.isPending ? '保存中...' : '保存'}
      </button>
      
      <button
        className="icon-btn"
        onClick={() => setShowSummaryInput(!showSummaryInput)}
        title="添加版本说明"
      >
        📝
      </button>
      
      {showSummaryInput && (
        <input
          type="text"
          value={summary}
          onChange={(e) => setSummary(e.target.value)}
          placeholder="版本说明（可选，AI会自动生成）"
        />
      )}
    </div>
  );
};
```
