# 签到系统

## 概述

签到系统是 YH-AI PPT 的**用户增长运营工具**，通过每日签到奖励机制提升用户活跃度和留存率。系统采用**阶梯式奖励设计**，连续签到可获得递增积分奖励，断签后重新累计，形成正向激励循环。

## 核心功能特性

| 功能特性 | 说明 | 技术亮点 |
|---------|------|---------|
| 每日签到 | 用户每日签到领取积分 | 幂等设计，防止重复签到 |
| 连续签到 | 连续签到获得递增奖励 | 阶梯算法，最多7天周期 |
| 签到日历 | 查看本月签到记录 | 可视化日历展示 |
| 断签重置 | 断签后连续天数归零 | 自动检测断签 |
| 积分发放 | 签到积分实时到账 | 事务保证 |
| 补签机制 | 支持积分补签 | 可选功能 |

## 数据库模型

### 1. 签到日志 (CheckInLog)

```prisma
model CheckInLog {
  id        String   @id @default(uuid())
  userId    String
  date      String   // 签到日期 (YYYY-MM-DD 格式)
  points    Int      // 本次签到获得积分
  streak    Int      // 连续签到天数
  createdAt DateTime @default(now())
  
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  @@unique([userId, date])  // 同一用户每天只能签到一次
  @@index([createdAt])
}
```

## 奖励机制算法

### 1. 阶梯奖励计算

```typescript
// server/src/services/growth.service.ts

/**
 * 计算签到奖励积分
 * 阶梯规则：基础5分 + 连续天数 - 1，最高11分
 */
function calculateCheckInPoints(streak: number): number {
  // 连续7天为一个周期
  const cycleDay = ((streak - 1) % 7) + 1;
  
  // 基础5分 + 递增
  const basePoints = 5;
  const bonus = cycleDay - 1;
  
  return Math.min(basePoints + bonus, 11);  // 最高11分
}

// 奖励对照表
const CHECKIN_REWARDS = {
  1: 5,   // 第1天：5积分
  2: 6,   // 第2天：6积分
  3: 7,   // 第3天：7积分
  4: 8,   // 第4天：8积分
  5: 9,   // 第5天：9积分
  6: 10,  // 第6天：10积分
  7: 11,  // 第7天：11积分（封顶）
  // 第8天开始循环
  8: 5,   // 第8天：回到5积分
  9: 6,   // 第9天：6积分
  // ...以此类推
};
```

### 2. 签到核心逻辑

```typescript
export async function checkIn(userId: string): Promise<{
  success: boolean;
  points: number;
  streak: number;
  totalPoints: number;
  message?: string;
}> {
  const today = new Date();
  const todayStr = today.toISOString().split('T')[0];  // YYYY-MM-DD
  
  // 1. 检查今日是否已签到（幂等检查）
  const existingCheckIn = await prisma.checkInLog.findUnique({
    where: {
      userId_date: {
        userId,
        date: todayStr
      }
    }
  });
  
  if (existingCheckIn) {
    return {
      success: false,
      points: 0,
      streak: 0,
      totalPoints: 0,
      message: '今日已签到，请明天再来'
    };
  }
  
  // 2. 获取用户当前连续签到状态
  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: { checkInDate: true, checkInStreak: true, points: true }
  });
  
  if (!user) {
    return {
      success: false,
      points: 0,
      streak: 0,
      totalPoints: 0,
      message: '用户不存在'
    };
  }
  
  // 3. 计算连续签到天数
  let newStreak = 1;  // 默认从1开始
  
  if (user.checkInDate) {
    const lastCheckIn = new Date(user.checkInDate);
    const yesterday = new Date(today);
    yesterday.setDate(yesterday.getDate() - 1);
    
    // 检查是否是连续签到（昨天签过）
    const isConsecutive = lastCheckIn.toISOString().split('T')[0] === yesterday.toISOString().split('T')[0];
    
    if (isConsecutive) {
      // 连续签到，累加
      newStreak = user.checkInStreak + 1;
    }
    // 如果不是连续（断签），newStreak保持为1，重新计数
  }
  
  // 4. 计算奖励积分
  const rewardPoints = calculateCheckInPoints(newStreak);
  
  // 5. 数据库事务：创建签到记录 + 更新用户积分
  await prisma.$transaction(async (tx) => {
    // 5.1 创建签到日志
    await tx.checkInLog.create({
      data: {
        userId,
        date: todayStr,
        points: rewardPoints,
        streak: newStreak
      }
    });
    
    // 5.2 更新用户签到状态和积分
    await tx.user.update({
      where: { id: userId },
      data: {
        checkInDate: today,
        checkInStreak: newStreak,
        points: { increment: rewardPoints }
      }
    });
    
    // 5.3 创建积分交易记录
    await tx.transaction.create({
      data: {
        userId,
        type: 'reward',
        amount: rewardPoints,
        balance: user.points + rewardPoints,
        description: `连续签到${newStreak}天奖励`,
        module: '增长运营',
        category: '签到'
      }
    });
  });
  
  return {
    success: true,
    points: rewardPoints,
    streak: newStreak,
    totalPoints: user.points + rewardPoints,
    message: `签到成功！获得${rewardPoints}积分，连续签到${newStreak}天`
  };
}
```

### 3. 断签检测逻辑

```typescript
/**
 * 检查用户是否断签，如果断签则重置连续天数
 * 此函数可放入定时任务每日执行
 */
export async function checkStreakReset(): Promise<void> {
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  const yesterdayStr = yesterday.toISOString().split('T')[0];
  
  // 找出所有连续签到状态非0，但昨天未签到的用户
  const usersToReset = await prisma.user.findMany({
    where: {
      checkInStreak: { gt: 0 },
      checkInDate: {
        lt: new Date(yesterdayStr)  // 上次签到早于昨天
      }
    }
  });
  
  for (const user of usersToReset) {
    await prisma.user.update({
      where: { id: user.id },
      data: { checkInStreak: 0 }
    });
    
    console.log(`[CheckIn] 用户 ${user.id} 断签，重置连续签到天数`);
  }
}

// 使用定时任务每天凌晨执行
cron.schedule('0 0 * * *', checkStreakReset);
```

## 前端集成

### 1. 签到按钮组件

```typescript
// src/components/CheckInButton.tsx

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';

export const CheckInButton: React.FC = () => {
  const queryClient = useQueryClient();
  
  // 获取签到状态
  const { data: status } = useQuery({
    queryKey: ['checkIn', 'status'],
    queryFn: async () => {
      const res = await fetch('/api/growth/check-in/status');
      return res.json();
    }
  });
  
  // 签到mutation
  const checkIn = useMutation({
    mutationFn: async () => {
      const res = await fetch('/api/growth/check-in', {
        method: 'POST'
      });
      return res.json();
    },
    onSuccess: (data) => {
      if (data.success) {
        toast.success(data.message);
        // 刷新状态
        queryClient.invalidateQueries({ queryKey: ['checkIn'] });
        queryClient.invalidateQueries({ queryKey: ['points', 'balance'] });
      } else {
        toast.error(data.message);
      }
    }
  });
  
  if (!status?.data) return null;
  
  const { hasCheckedInToday, streak, todayPoints } = status.data;
  
  return (
    <div className="check-in-button">
      <button
        onClick={() => checkIn.mutate()}
        disabled={hasCheckedInToday || checkIn.isPending}
        className={hasCheckedInToday ? 'checked' : ''}
      >
        {hasCheckedInToday ? (
          '今日已签到'
        ) : checkIn.isPending ? (
          '签到中...'
        ) : (
          `签到 +${todayPoints}积分`
        )}
      </button>
      
      <div className="streak-info">
        连续签到 {streak} 天
      </div>
    </div>
  );
};
```

### 2. 签到日历组件

```typescript
// src/components/CheckInCalendar.tsx

import { useState } from 'react';

export const CheckInCalendar: React.FC<{
  calendar: Array<{
    date: string;
    checkedIn: boolean;
    points: number;
  }>;
}> = ({ calendar }) => {
  const [currentMonth, setCurrentMonth] = useState(new Date());
  
  // 生成日历网格
  const generateCalendarDays = () => {
    const year = currentMonth.getFullYear();
    const month = currentMonth.getMonth();
    
    const firstDay = new Date(year, month, 1);
    const lastDay = new Date(year, month + 1, 0);
    
    const days = [];
    const startPadding = firstDay.getDay();
    
    // 前填充
    for (let i = 0; i < startPadding; i++) {
      days.push(null);
    }
    
    // 日期
    for (let i = 1; i <= lastDay.getDate(); i++) {
      const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(i).padStart(2, '0')}`;
      const record = calendar.find(c => c.date === dateStr);
      
      days.push({
        day: i,
        date: dateStr,
        checkedIn: record?.checkedIn || false,
        points: record?.points || 0
      });
    }
    
    return days;
  };
  
  const days = generateCalendarDays();
  
  return (
    <div className="check-in-calendar">
      <div className="calendar-header">
        <button onClick={() => setCurrentMonth(new Date(currentMonth.setMonth(currentMonth.getMonth() - 1)))}>
          上月
        </button>
        <span>{currentMonth.getFullYear()}年{currentMonth.getMonth() + 1}月</span>
        <button onClick={() => setCurrentMonth(new Date(currentMonth.setMonth(currentMonth.getMonth() + 1)))}>
          下月
        </button>
      </div>
      
      <div className="calendar-grid">
        {['日', '一', '二', '三', '四', '五', '六'].map(d => (
          <div key={d} className="weekday">{d}</div>
        ))}
        
        {days.map((day, i) => (
          <div
            key={i}
            className={`calendar-day ${day?.checkedIn ? 'checked' : ''} ${
              day?.date === new Date().toISOString().split('T')[0] ? 'today' : ''
            }`}
          >
            {day?.day || ''}
            {day?.checkedIn && <span className="points">+{day.points}</span>}
          </div>
        ))}
      </div>
    </div>
  );
};
```

### 3. 补签机制

```typescript
// 补签功能（可选）
export async function makeUpCheckIn(
  userId: string,
  date: string,  // YYYY-MM-DD
  costPoints: number = 10
): Promise<boolean> {
  // 检查积分是否充足
  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: { points: true }
  });
  
  if (!user || user.points < costPoints) {
    throw new Error('积分不足');
  }
  
  // 扣除积分并创建补签记录
  await prisma.$transaction([
    prisma.user.update({
      where: { id: userId },
      data: { points: { decrement: costPoints } }
    }),
    prisma.checkInLog.create({
      data: {
        userId,
        date,
        points: 0,  // 补签不获得积分
        streak: 0,  // 补签不计入连续
        isMakeUp: true
      }
    })
  ]);
  
  return true;
}
```
