# 系统设置与配置

## 概述

系统设置与配置模块负责管理平台运行时配置，支持从环境变量加载默认配置、数据库存储用户自定义配置、以及配置热更新（无需重启服务即可生效）。系统提供配置掩码功能保护API密钥等敏感信息，同时支持配置的导入导出和版本管理。

## 核心功能

| 功能 | 说明 | 实现位置 |
|------|------|----------|
| 环境变量加载 | 从.env文件加载默认配置 | SettingService.loadFromEnv() |
| 数据库配置 | AppSettings表存储用户配置 | Prisma AppSettings模型 |
| 配置热更新 | 修改后立即生效 | SettingService.updateSettings() |
| 配置掩码 | API Key等敏感信息脱码显示 | maskApiKey(), maskKeysInObject() |
| 配置重置 | 恢复为环境变量默认值 | SettingService.resetToEnv() |
| 环境预设 | 各提供商默认配置模板 | SettingService.getEnvPresets() |

## 核心实现

### 设置服务完整代码

```typescript
// server/src/services/setting.service.ts

import { prisma } from '../db';

export class SettingService {
  // API密钥脱敏（仅显示后4位）
  private static maskApiKey(key: string | undefined): string {
    if (!key || key.length < 8) return key ? '••••••••' : '';
    return `••••••••${key.slice(-4)}`;
  }

  // 递归脱敏对象中的所有apiKey字段
  private static maskKeysInObject(obj: any): any {
    if (!obj || typeof obj !== 'object') return obj;
    const result: any = Array.isArray(obj) ? [] : {};

    for (const key of Object.keys(obj)) {
      if (key.toLowerCase() === 'apikey' && typeof obj[key] === 'string') {
        result[key] = this.maskApiKey(obj[key]);
      } else if (typeof obj[key] === 'object') {
        result[key] = this.maskKeysInObject(obj[key]);
      } else {
        result[key] = obj[key];
      }
    }
    return result;
  }

  // 获取完整配置（后端内部使用）
  static async getSettings() {
    const settings = await (prisma as any).appSettings.findFirst({
      where: { id: 'global' }
    });

    if (!settings) return null;

    try {
      return JSON.parse(settings.config);
    } catch {
      return null;
    }
  }

  // 获取脱敏配置（前端展示用）
  static async getMaskedSettings() {
    const fullSettings = await this.getSettings();
    if (!fullSettings) return null;
    return this.maskKeysInObject(fullSettings);
  }

  // 从环境变量加载配置
  static loadFromEnv(): any {
    const env = process.env;
    const getEnv = (key: string) => env[key] || '';
    const getEnvNum = (key: string) => {
      const val = parseInt(env[key] || '');
      return isNaN(val) ? undefined : val;
    };

    const provider = getEnv('AI_PROVIDER') || 'Gemini';

    const prefixes: Record<string, string> = {
      'Gemini': 'GEMINI',
      'OpenAI': 'OPENAI',
      'Zhipu': 'ZHIPU',
      'SiliconFlow': 'SILICON',
      'ModelScope': 'MODELSCOPE',
      'Volcengine': 'VOLCENGINE',
      'Custom': 'CUSTOM'
    };

    const prefix = prefixes[provider];

    const getConfig = (specificKey: string, fallbackKey: string, defaultValue: string) => {
      if (prefix) {
        const specific = getEnv(`${prefix}_${specificKey}`);
        if (specific) return specific;
      }
      const generic = getEnv(fallbackKey);
      if (generic) return generic;
      return defaultValue;
    };

    const defaults: Record<string, any> = {
      'Gemini': {
        base: 'https://generativelanguage.googleapis.com',
        text: 'gemini-3-pro-preview',
        image: 'gemini-3-pro-image',
        vision: 'gemini-3-pro-preview'
      },
      'OpenAI': {
        base: 'https://api.openai.com/v1',
        text: 'gpt-4-turbo',
        image: 'dall-e-3',
        vision: 'gpt-4-vision-preview'
      },
      'Zhipu': {
        base: 'https://open.bigmodel.cn/api/paas/v4',
        text: 'glm-4',
        image: 'cogview-3',
        vision: 'glm-4v'
      },
      'Volcengine': {
        base: 'https://ark.cn-beijing.volces.com/api/v3',
        text: 'doubao-pro-256k',
        image: 'doubao-image',
        vision: 'doubao-vision-pro'
      }
    };

    const preset = defaults[provider] || defaults['Gemini'];

    return {
      ai: {
        provider,
        baseUrl: getConfig('BASE_URL', 'AI_BASE_URL', preset.base),
        apiKey: getConfig('API_KEY', 'AI_API_KEY', ''),
        models: {
          text: getConfig('MODEL_TEXT', 'AI_MODEL_TEXT', preset.text),
          image: getConfig('MODEL_IMAGE', 'AI_MODEL_IMAGE', preset.image),
          vision: getConfig('MODEL_VISION', 'AI_MODEL_VISION', preset.vision)
        },
        customCombo: {
          text: {
            baseUrl: getEnv('COMBO_TEXT_BASE') || '',
            apiKey: getEnv('COMBO_TEXT_KEY') || '',
            model: getEnv('COMBO_TEXT_MODEL') || ''
          },
          image: {
            baseUrl: getEnv('COMBO_IMAGE_BASE') || '',
            apiKey: getEnv('COMBO_IMAGE_KEY') || '',
            model: getEnv('COMBO_IMAGE_MODEL') || ''
          },
          vision: {
            baseUrl: getEnv('COMBO_VISION_BASE') || '',
            apiKey: getEnv('COMBO_VISION_KEY') || '',
            model: getEnv('COMBO_VISION_MODEL') || ''
          }
        }
      },
      docParser: {
        provider: getEnv('DOC_PARSER_PROVIDER') || 'MinerU',
        baseUrl: getEnv('DOC_PARSER_BASE') || 'https://mineru.net',
        apiKey: getEnv('DOC_PARSER_KEY') || ''
      },
      imageGeneration: {
        resolution: getEnv('IMG_RESOLUTION') || '2048x2048'
      },
      performance: {
        textConcurrency: getEnvNum('PERF_TEXT_CONCURRENCY') || 1,
        imageConcurrency: getEnvNum('PERF_IMAGE_CONCURRENCY') || 1
      },
      language: getEnv('OUTPUT_LANGUAGE') || 'zh'
    };
  }

  // 重置为环境变量默认值
  static async resetToEnv() {
    const envSettings = this.loadFromEnv();
    return await this.updateSettings(envSettings);
  }

  // 获取环境预设
  static getEnvPresets() {
    const env = process.env;
    const getEnv = (key: string) => env[key] || '';

    const prefixMap: Record<string, string> = {
      'Gemini': 'GEMINI',
      'OpenAI': 'OPENAI',
      'Zhipu': 'ZHIPU',
      'SiliconFlow': 'SILICON',
      'ModelScope': 'MODELSCOPE',
      'Volcengine': 'VOLCENGINE'
    };

    const defaults = {
      'Gemini': {
        base: 'https://generativelanguage.googleapis.com',
        text: 'gemini-3-pro-preview',
        image: 'gemini-3-pro-image',
        vision: 'gemini-3-pro-preview'
      },
      'OpenAI': {
        base: 'https://api.openai.com/v1',
        text: 'gpt-4-turbo',
        image: 'dall-e-3',
        vision: 'gpt-4-vision-preview'
      }
    };

    const presets: Record<string, any> = {};

    for (const [provider, prefix] of Object.entries(prefixMap)) {
      const defaultPreset = (defaults as any)[provider];
      presets[provider] = {
        baseUrl: getEnv(`${prefix}_BASE_URL`) || defaultPreset?.base || '',
        models: {
          text: getEnv(`${prefix}_MODEL_TEXT`) || defaultPreset?.text || '',
          image: getEnv(`${prefix}_MODEL_IMAGE`) || defaultPreset?.image || '',
          vision: getEnv(`${prefix}_MODEL_VISION`) || defaultPreset?.vision || ''
        }
      };
    }
    return presets;
  }

  // 启动时同步环境变量到数据库
  static async syncEnvToDatabase() {
    const existingSettings = await this.getSettings();
    if (!existingSettings) {
      console.log('[SettingService] No settings found. Initializing with .env defaults...');
      const envSettings = this.loadFromEnv();
      await this.updateSettings(envSettings);
    } else {
      console.log('[SettingService] Settings already exist. Skipping .env overwrite.');
    }
  }

  // 更新配置（自动处理脱敏字段）
  static async updateSettings(config: any) {
    const currentSettings = await this.getSettings();

    // 递归恢复脱敏的密钥
    const restoreKeys = (incoming: any, original: any): any => {
      if (!incoming || typeof incoming !== 'object' || !original) {
        return incoming;
      }

      if (Array.isArray(incoming)) return incoming;

      const result = { ...incoming };

      for (const key in result) {
        const val = result[key];
        const originalVal = original[key];

        // 如果值以••••开头，说明是被脱敏的密钥
        if (typeof val === 'string' && val.startsWith('••••')) {
          if (originalVal) {
            result[key] = originalVal;
          }
        } else if (typeof val === 'object' && val !== null) {
          result[key] = restoreKeys(val, originalVal);
        }
      }
      return result;
    };

    const finalConfig = restoreKeys(config, currentSettings);
    const configStr = JSON.stringify(finalConfig);

    return await (prisma as any).appSettings.upsert({
      where: { id: 'global' },
      create: {
        id: 'global',
        config: configStr
      },
      update: {
        config: configStr
      }
    });
  }

  // 热重载.env文件
  static async reloadEnv() {
    try {
      const envPath = require('path').resolve(process.cwd(), '.env');
      const fs = require('fs');
      const dotenv = require('dotenv');

      if (fs.existsSync(envPath)) {
        console.log('[SettingService] Reloading .env from:', envPath);
        const envConfig = dotenv.parse(fs.readFileSync(envPath));

        // 更新process.env
        for (const k in envConfig) {
          process.env[k] = envConfig[k];
        }

        // 同步到数据库
        await this.syncEnvToDatabase();
      }
    } catch (error) {
      console.error('[SettingService] Failed to hot-reload .env:', error);
    }
  }
}
```

## API 接口

### 路由配置

```typescript
// server/src/routes/setting.routes.ts

import { Router } from 'express';
import {
  handleGetSettings,
  handleGetMaskedSettings,
  handleUpdateSettings,
  handleResetSettings
} from '../controllers/setting.controller';
import {
  authenticate,
  requireAdmin,
  requirePermission
} from '../middlewares/auth.middleware';

const router = Router();

// GET /api/settings - 完整配置（后端服务使用）
router.get('/', handleGetSettings);

// GET /api/settings/masked - 脱敏配置（前端展示用）
router.get('/masked',
  authenticate,
  requireAdmin,
  requirePermission('admin.settings.view'),
  handleGetMaskedSettings
);

// POST /api/settings - 更新配置
router.post('/',
  authenticate,
  requireAdmin,
  requirePermission('admin.settings.update.general'),
  handleUpdateSettings
);

// POST /api/settings/reset - 重置为环境变量默认值
router.post('/reset',
  authenticate,
  requireAdmin,
  requirePermission('admin.settings.reset'),
  handleResetSettings
);

export default router;
```

### 控制器实现

```typescript
// server/src/controllers/setting.controller.ts

import { Request, Response } from 'express';
import { SettingService } from '../services/setting.service';

// 返回完整配置（用于后端服务内部调用）
export const handleGetSettings = async (req: Request, res: Response) => {
  try {
    const settings = await SettingService.getSettings();
    res.json({ success: true, data: settings });
  } catch (error: any) {
    res.status(500).json({ success: false, error: error.message });
  }
};

// 返回脱敏配置（前端展示用）
export const handleGetMaskedSettings = async (req: Request, res: Response) => {
  try {
    const settings = await SettingService.getMaskedSettings();
    const envPresets = SettingService.getEnvPresets();
    res.json({ success: true, data: settings, envPresets });
  } catch (error: any) {
    res.status(500).json({ success: false, error: error.message });
  }
};

// 更新配置
export const handleUpdateSettings = async (req: Request, res: Response) => {
  try {
    const { config } = req.body;
    await SettingService.updateSettings(config);
    res.json({ success: true, message: 'Settings updated successfully' });
  } catch (error: any) {
    res.status(500).json({ success: false, error: error.message });
  }
};

// 重置配置
export const handleResetSettings = async (req: Request, res: Response) => {
  try {
    await SettingService.resetToEnv();
    res.json({
      success: true,
      message: 'Settings reset to environment defaults'
    });
  } catch (error: any) {
    res.status(500).json({ success: false, error: error.message });
  }
};
```

## 前端组件

### 系统设置页面

```tsx
// SystemSettings.tsx
import React, { useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { toast } from 'react-hot-toast';
import { Settings, Save, RotateCcw } from 'lucide-react';
import { AIEngineConfig } from './AIEngineConfig';
import { DocParserConfig } from './DocParserConfig';
import { PerformanceConfig } from './PerformanceConfig';

const TABS = [
  { id: 'ai', label: 'AI引擎', component: AIEngineConfig },
  { id: 'doc', label: '文档解析', component: DocParserConfig },
  { id: 'perf', label: '性能配置', component: PerformanceConfig }
];

export const SystemSettings: React.FC = () => {
  const [activeTab, setActiveTab] = useState('ai');
  const [formData, setFormData] = useState<any>({});

  const { data: settings } = useQuery({
    queryKey: ['settings'],
    queryFn: fetchMaskedSettings
  });

  const updateMutation = useMutation({
    mutationFn: updateSettings,
    onSuccess: () => toast.success('配置已保存'),
    onError: () => toast.error('保存失败')
  });

  const resetMutation = useMutation({
    mutationFn: resetSettings,
    onSuccess: () => {
      toast.success('已重置为默认值');
      window.location.reload();
    }
  });

  const ActiveComponent = TABS.find(t => t.id === activeTab)?.component;

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <h2 className="text-2xl font-bold flex items-center gap-2">
          <Settings className="w-6 h-6" />
          系统设置
        </h2>
        <div className="flex gap-2">
          <button
            onClick={() => resetMutation.mutate()}
            disabled={resetMutation.isPending}
            className="flex items-center gap-2 px-4 py-2 border rounded-lg hover:bg-gray-50"
          >
            <RotateCcw className="w-4 h-4" />
            重置
          </button>
          <button
            onClick={() => updateMutation.mutate({ config: formData })}
            disabled={updateMutation.isPending}
            className="flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700"
          >
            <Save className="w-4 h-4" />
            保存
          </button>
        </div>
      </div>

      {/* 标签页 */}
      <div className="border-b">
        <div className="flex gap-4">
          {TABS.map(tab => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id)}
              className={`px-4 py-2 border-b-2 transition-colors ${
                activeTab === tab.id
                  ? 'border-blue-500 text-blue-600'
                  : 'border-transparent hover:border-gray-300'
              }`}
            >
              {tab.label}
            </button>
          ))}
        </div>
      </div>

      {/* 配置表单 */}
      {ActiveComponent && (
        <ActiveComponent
          settings={settings?.data}
          onChange={setFormData}
        />
      )}
    </div>
  );
};
```

## 配置热更新流程

```
1. 管理员在前端修改配置
   ↓
2. 调用 POST /api/settings
   ↓
3. 后端自动处理脱敏字段恢复
   ↓
4. 更新数据库 AppSettings 表
   ↓
5. 立即生效（下次读取时从数据库获取）
   ↓
6. AI服务自动使用新配置
```
