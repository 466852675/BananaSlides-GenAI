# 管理后台概览

## 概述

管理后台是 YH-AI PPT 系统的运营管理中枢，为管理员提供完整的数据可视化、业务运营和系统管理能力。后台采用模块化设计，包含数据概览、用户管理、订单管理、积分管理、退款审核、线索管理、权限管理、AI引擎配置和系统设置等核心模块。

// DashboardStats.tsx
import React from 'react';
import { useQuery } from '@tanstack/react-query';
import { Users, ShoppingCart, Coins, Activity } from 'lucide-react';

interface StatCardProps {
  title: string;
  value: string | number;
  change?: number;
  icon: React.ReactNode;
  color: string;
}

const StatCard: React.FC<StatCardProps> = ({ title, value, change, icon, color }) => (
  <div className="bg-white p-6 rounded-lg shadow-sm border border-gray-100">
    <div className="flex items-center justify-between">
      <div>
        <p className="text-sm text-gray-500">{title}</p>
        <p className="text-2xl font-bold mt-1">{value}</p>
        {change !== undefined && (
          <p className={`text-sm mt-2 ${change >= 0 ? 'text-green-600' : 'text-red-600'}`}>
            {change >= 0 ? '+' : ''}{change}% 较上月
          </p>
        )}
      </div>
      <div className={`p-3 rounded-lg ${color}`}>
        {icon}
      </div>
    </div>
  </div>
);

export const DashboardStats: React.FC = () => {
  const { data: stats } = useQuery({
    queryKey: ['dashboard-stats'],
    queryFn: fetchDashboardStats
  });

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
      <StatCard
        title="总用户数"
        value={stats?.totalUsers?.toLocaleString() || '-'}
        change={stats?.userGrowth}
        icon={<Users className="w-6 h-6 text-blue-600" />}
        color="bg-blue-50"
      />
      <StatCard
        title="今日订单"
        value={stats?.todayOrders?.toLocaleString() || '-'}
        change={stats?.orderGrowth}
        icon={<ShoppingCart className="w-6 h-6 text-green-600" />}
        color="bg-green-50"
      />
      <StatCard
        title="今日收入"
        value={`¥${stats?.todayRevenue?.toLocaleString() || '-'}`}

## 后端服务实现

### Admin 统计服务

```typescript
// server/src/services/admin.service.ts

import { prisma } from '../db';
import { startOfDay, endOfDay, subDays } from 'date-fns';

export class AdminService {
  /**
   * 获取Dashboard统计数据
   */
  static async getDashboardStats() {
    const today = new Date();
    const yesterday = subDays(today, 1);
    const lastMonth = subDays(today, 30);

    // 并行查询各项统计
    const [
      totalUsers,
      todayUsers,
      yesterdayUsers,
      lastMonthUsers,
      todayOrders,
      yesterdayOrders,
      todayRevenue,
      yesterdayRevenue,
      activeProjects
    ] = await Promise.all([
      // 总用户数
      prisma.user.count(),
      
      // 今日新增用户
      prisma.user.count({
        where: {
          createdAt: {
            gte: startOfDay(today),
            lte: endOfDay(today)
          }
        }
      }),
      
      // 昨日新增用户
      prisma.user.count({
        where: {
          createdAt: {
            gte: startOfDay(yesterday),
            lte: endOfDay(yesterday)
          }
        }
      }),
      
      // 上月用户数
      prisma.user.count({
        where: {
          createdAt: { lte: lastMonth }
        }
      }),
      
      // 今日订单
      prisma.order.count({
        where: {
          createdAt: {
            gte: startOfDay(today),
            lte: endOfDay(today)
          }
        }
      }),
      
      // 昨日订单
      prisma.order.count({
        where: {
          createdAt: {
            gte: startOfDay(yesterday),
            lte: endOfDay(yesterday)
          }
        }
      }),
      
      // 今日收入
      prisma.order.aggregate({
        where: {
          createdAt: {
            gte: startOfDay(today),
            lte: endOfDay(today)
          },
          status: 'PAID'
        },
        _sum: { finalPrice: true }
      }),
      
      // 昨日收入
      prisma.order.aggregate({
        where: {
          createdAt: {
            gte: startOfDay(yesterday),
            lte: endOfDay(yesterday)
          },
          status: 'PAID'
        },
        _sum: { finalPrice: true }
      }),
      
      // 活跃项目数
      prisma.project.count({
        where: {
          updatedAt: {
            gte: subDays(today, 7)
          }
        }
      })
    ]);

    // 计算增长率
    const userGrowth = yesterdayUsers > 0
      ? ((todayUsers - yesterdayUsers) / yesterdayUsers * 100).toFixed(1)
      : 0;
    
    const orderGrowth = yesterdayOrders > 0
      ? ((todayOrders - yesterdayOrders) / yesterdayOrders * 100).toFixed(1)
      : 0;
    
    const revenueGrowth = yesterdayRevenue._sum.finalPrice > 0
      ? ((todayRevenue._sum.finalPrice - yesterdayRevenue._sum.finalPrice) / yesterdayRevenue._sum.finalPrice * 100).toFixed(1)
      : 0;

    return {
      totalUsers,
      todayUsers,
      userGrowth: Number(userGrowth),
      todayOrders,
      orderGrowth: Number(orderGrowth),
      todayRevenue: todayRevenue._sum.finalPrice || 0,
      revenueGrowth: Number(revenueGrowth),
      activeProjects
    };
  }

  /**
   * 获取趋势数据
   */
  static async getTrends(days: number = 30) {
    const dates = Array.from({ length: days }, (_, i) =>
      subDays(new Date(), days - i - 1)
    );

    const userTrends = await Promise.all(
      dates.map(async date => ({
        date: date.toISOString().split('T')[0],
        count: await prisma.user.count({
          where: {
            createdAt: {
              gte: startOfDay(date),
              lte: endOfDay(date)
            }
          }
        })
      }))
    );

    const orderTrends = await Promise.all(
      dates.map(async date => ({
        date: date.toISOString().split('T')[0],
        count: await prisma.order.count({
          where: {
            createdAt: {
              gte: startOfDay(date),
              lte: endOfDay(date)
            }
          }
        })
      }))
    );

    return {
      users: userTrends,
      orders: orderTrends
    };
  }
}
```
